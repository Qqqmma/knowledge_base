## 📊 Статистика года

```dataviewjs
const currentYear = dv.date("today").year;
const movies = dv.pages("")
    .where(p => p.type === "movie" && p.status === "Посмотрен")
    .where(p => p.date_watched && dv.date(p.date_watched).year === currentYear);

const count = movies.length;
const avgRating = count > 0 
    ? (movies.array().reduce((sum, m) => sum + (Number(m.rating) || 0), 0) / count).toFixed(1)
    : "—";

const genres = {};
movies.forEach(m => {
    const g = Array.isArray(m.genre) ? m.genre : [m.genre];
    g.forEach(genre => {
        if (genre) genres[genre] = (genres[genre] || 0) + 1;
    });
});
const topGenre = Object.entries(genres).sort((a, b) => b[1] - a[1])[0];

dv.paragraph(`🎬 **Посмотрено в ${currentYear}:** ${count} фильмов`);
dv.paragraph(`⭐ **Средняя оценка:** ${avgRating}/10`);
if (topGenre) dv.paragraph(`🎭 **Любимый жанр:** ${topGenre[0]} (${topGenre[1]} шт.)`);
```

    year as "Год"

## ✅ Посмотрено в этом году

```dataview
TABLE WITHOUT ID
    file.link as "Фильм",
    join(director, ", ") as "Режиссёр",
    join(genre, ", ") as "Жанр",
    rating as "Оценка",
    dateformat(date_watched, "dd.MM") as "Дата"
WHERE type = "movie" AND status = "Посмотрен" AND dateformat(date_watched, "yyyy") = dateformat(date(today), "yyyy")
SORT date_watched DESC
```

## 🎲 Случайный фильм на вечер

```dataviewjs
const movies = dv.pages("")
    .where(p => p.type === "movie" && p.status === "Планирую");
if (movies.length > 0) {
    const random = movies[Math.floor(Math.random() * movies.length)];
    const dir = Array.isArray(random.director) ? random.director.join(", ") : random.director;
    const gen = Array.isArray(random.genre) ? random.genre.join(", ") : random.genre;
    dv.paragraph(`🎬 **${random.file.link}** — ${dir} (${random.year})`);
    dv.paragraph(`🎭 ${gen}`);
} else {
    dv.paragraph("*Список планов пуст. Пора добавить что-нибудь!*");
}
```
